how to take an input in C++ and assign it to a value
where you don't know the data type

Code example 


#include <iostream>
#include <string>
#include <variant>
#include <sstream>
#include <cctype> // Good practice to include for std::isdigit

using DynamicValue = std::variant<int, double, char, std::string>;

DynamicValue parseInput(const std::string& input) {
    if (input.length() == 1 && !std::isdigit(static_cast<unsigned char>(input[0]))) {
        return input[0];
    }

    std::istringstream intStream(input);
    int intVal;
    if ((intStream >> intVal) && (intStream >> std::ws).eof()) {
        return intVal;
    }

    std::istringstream doubleStream(input);
    double doubleVal;
    if ((doubleStream >> doubleVal) && (doubleStream >> std::ws).eof()) {
        return doubleVal;
    }

    return input;
}

int main() {
    std::string rawInput;
    std::cout << "Enter anything: ";
    std::getline(std::cin, rawInput);

    DynamicValue value = parseInput(rawInput);

    // Check which type was stored and extract it
    if (std::holds_alternative<int>(value)) {
        int myInt = std::get<int>(value);
        std::cout << "It is an int: " << myInt << '\n';
    } 
    else if (std::holds_alternative<double>(value)) {
        double myDouble = std::get<double>(value);
        std::cout << "It is a double: " << myDouble << '\n';
    } 
    else if (std::holds_alternative<char>(value)) {
        char myChar = std::get<char>(value);
        std::cout << "It is a char: " << myChar << '\n';
    } 
    else if (std::holds_alternative<std::string>(value)) {
        std::string myString = std::get<std::string>(value);
        std::cout << "It is a string: " << myString << '\n';
    }

    return 0;
}

    int numerator = 10;
    int denominator = 0;

    if (denominator != 0) {
        int result = numerator / denominator;
        std::cout << "Result: " << result << "\n";
    } else {
        std::cerr << "Error: Cannot divide integers by zero!\n";
        // Handle error gracefully (e.g., return, assign a default value, or exit)
    }
}
